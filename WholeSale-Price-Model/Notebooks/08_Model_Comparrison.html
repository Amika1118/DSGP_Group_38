def rmse(y_true, y_pred):
    return np.sqrt(mean_squared_error(y_true, y_pred))

def evaluate_model(model, X_tr, y_tr, X_te, y_te, name, model_type):
    pred_tr = model.predict(X_tr)
    pred_te = model.predict(X_te)
    train_r2 = r2_score(y_tr, pred_tr)
    test_r2 = r2_score(y_te, pred_te)
    train_rmse = rmse(y_tr, pred_tr)
    test_rmse = rmse(y_te, pred_te)
    train_mae = mean_absolute_error(y_tr, pred_tr)
    test_mae = mean_absolute_error(y_te, pred_te)
    gap = train_r2 - test_r2
    return {
        'Model': name,
        'Type': model_type,
        'Train R2': train_r2,
        'Test R2': test_r2,
        'Train RMSE': train_rmse,
        'Test RMSE': test_rmse,
        'Train MAE': train_mae,
        'Test MAE': test_mae,
        'Gap': gap
    }

def judge(row):
    if row['Test R2'] < 0:
        return "Test R2 negative – severe concept drift"
    if row['Gap'] < 0.1:
        return "Excellent – very low overfitting"
    elif row['Gap'] < 0.2:
        return "Good – mild overfitting"
    elif row['Gap'] < 0.3:
        return "Moderate overfitting – acceptable"
    else:
        return "Severe overfitting – needs more regularisation"

results_df['Judgment'] = results_df.apply(judge, axis=1)
